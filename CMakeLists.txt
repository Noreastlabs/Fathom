option(FATHOM_BUILD_DESKTOP "Build the complete private macOS desktop application" ON)
if(FATHOM_BUILD_DESKTOP)
    if(NOT APPLE)
        message(FATAL_ERROR "The desktop implementation currently requires macOS")
    endif()
    include(cmake/Desktop.cmake)
    add_custom_command(TARGET Fathom POST_BUILD
        COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/Tools/stage_runtime_assets.py"

        VERBATIM)
endif()
